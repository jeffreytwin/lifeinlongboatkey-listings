export async function bulkInsertToStagingPP(housesData) {
    const chunkSize = 50; // set the chunk size
    const chunks = housesData.reduce((acc, _, i) => (i % chunkSize ? acc : [...acc, housesData.slice(i, i + chunkSize)]), []);

    let insertedIds = [];
    for (const chunk of chunks) {
        try {
            const result = await wixData.bulkInsert('Stagging', chunk);
            insertedIds = insertedIds.concat(result.insertedItemIds);
        } catch (error) {
            console.log('Error on bulkInsertToStaging() ->', error);
        }
    }
    return insertedIds;
}

export function getNumber(number) {
    let isNUm = typeof (number)
    if (isNUm === "number") {
        number = number.toString()
        if (number.length === 6) {
            let shortNum = `$${number.slice(0, 1)   }00s`
            return shortNum
        } else if (number.length >= 7 && number.length <= 9) {
            let numLen = number.length
            let shortNum = numLen === 7 ? number.slice(0, 1) : numLen === 8 ? number.slice(0, 2) : numLen === 9 && number.slice(0, 3)
            return `${shortNum}M+`
        } else {
            return `$${number}`
        }
    } else {
        return isNUm
    }
}

export async function updateInsertDate() {
    const dateOfPull = new Date();

    try {
        // Fetch all items from the "HousesforSale" collection
        const results = await wixData.query("HousesforSale").limit(1000).find();

        // Use Promise.all to update all items in parallel
        await Promise.all(results.items.map(async (element) => {
            try {
                const vax = { ...element };
                vax.dateOfMlsPull = dateOfPull;

                // Use wixData.update to update each item
                await wixData.update('HousesforSale', vax);

                console.log('- - - Data Updated Successfully - - -');
            } catch (e) {
                console.log(`Error on update for item with _id ${element._id} - - >`, e);
            }
        }));

        console.log('All data updated successfully.');
        return "done";
    } catch (e) {
        console.log('Error on fetching or updating data in updateInsertDate - - >', e);
        return "error";
    }
}
